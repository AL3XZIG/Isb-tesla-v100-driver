option(ISB_BUILD_CONTROL_CENTER "Build the optional Qt 6 Widgets Control Center" ON)
if(ISB_BUILD_CONTROL_CENTER)
  find_package(Qt6 COMPONENTS Widgets QUIET)
  if(Qt6Widgets_FOUND)
    qt_standard_project_setup()
    qt_add_executable(isb-v100-control-center qt/main.cpp)
    target_link_libraries(isb-v100-control-center PRIVATE isb-control-center-core Qt6::Widgets)
    target_compile_features(isb-v100-control-center PRIVATE cxx_std_17)
  else()
    message(STATUS "Qt 6 Widgets not found; isb-v100-control-center target is disabled")
  endif()
endif()

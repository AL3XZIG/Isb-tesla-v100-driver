add_library(isb-nvml src/nvml_provider.cpp src/hub_provider.cpp)
target_include_directories(isb-nvml PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(isb-nvml PUBLIC isb-common PRIVATE isb-hub)
target_compile_features(isb-nvml PUBLIC cxx_std_17)

find_path(ISB_NVML_INCLUDE_DIR NAMES nvml.h)
find_library(ISB_NVML_LIBRARY NAMES nvidia-ml nvml)
if(ISB_NVML_INCLUDE_DIR AND ISB_NVML_LIBRARY)
    target_include_directories(isb-nvml PRIVATE ${ISB_NVML_INCLUDE_DIR})
    target_link_libraries(isb-nvml PRIVATE ${ISB_NVML_LIBRARY})
    target_compile_definitions(isb-nvml PUBLIC ISB_HAS_NVML=1)
    message(STATUS "NVML support enabled")
else()
    message(STATUS "NVML support disabled: nvml.h and libnvidia-ml were not both found")
endif()

add_library(isb-cuda src/cuda_provider.cpp)
target_include_directories(isb-cuda PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(isb-cuda PUBLIC isb-common)
target_compile_features(isb-cuda PUBLIC cxx_std_17)

find_path(ISB_CUDA_INCLUDE_DIR NAMES cuda_runtime_api.h)
find_library(ISB_CUDA_RUNTIME_LIBRARY NAMES cudart)
if(ISB_CUDA_INCLUDE_DIR AND ISB_CUDA_RUNTIME_LIBRARY)
    target_include_directories(isb-cuda PRIVATE ${ISB_CUDA_INCLUDE_DIR})
    target_link_libraries(isb-cuda PRIVATE ${ISB_CUDA_RUNTIME_LIBRARY})
    target_compile_definitions(isb-cuda PRIVATE ISB_HAS_CUDA_RUNTIME=1)
    set(ISB_CUDA_RUNTIME_FOUND TRUE)
    message(STATUS "CUDA Runtime provider enabled")
else()
    set(ISB_CUDA_RUNTIME_FOUND FALSE)
    message(STATUS "CUDA Runtime provider disabled: cuda_runtime_api.h and cudart were not both found")
endif()

if(BUILD_TESTING)
    add_executable(isb-cuda-tests tests/provider_contract_test.cpp)
    target_link_libraries(isb-cuda-tests PRIVATE isb-cuda)
    if(ISB_CUDA_RUNTIME_FOUND)
        target_compile_definitions(isb-cuda-tests PRIVATE ISB_HAS_CUDA_RUNTIME=1)
    endif()
    add_test(NAME isb-cuda-tests COMMAND isb-cuda-tests)
endif()

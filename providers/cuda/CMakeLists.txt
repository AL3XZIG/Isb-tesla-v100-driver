add_library(isb-cuda src/cuda_provider.cpp)
target_include_directories(isb-cuda PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(isb-cuda PUBLIC isb-common)
target_compile_features(isb-cuda PUBLIC cxx_std_17)

find_package(CUDAToolkit QUIET)
if(CUDAToolkit_FOUND)
    target_link_libraries(isb-cuda PRIVATE CUDA::cudart)
    target_compile_definitions(isb-cuda PRIVATE ISB_HAS_CUDA_RUNTIME=1)
    message(STATUS "CUDA Runtime provider enabled")
else()
    message(STATUS "CUDA Runtime provider disabled: CUDAToolkit not found")
endif()

if(BUILD_TESTING)
    add_executable(isb-cuda-tests tests/provider_contract_test.cpp)
    target_link_libraries(isb-cuda-tests PRIVATE isb-cuda)
    add_test(NAME isb-cuda-tests COMMAND isb-cuda-tests)
endif()
